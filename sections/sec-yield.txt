<emu-clause id="sec-yield" type="abstract operation">
        <h1>
          Yield (
            _arg_: an ECMAScript language value,
          ): either a normal completion containing an ECMAScript language value or an abrupt completion
        </h1>
        <dl class="header">
        </dl>
        <emu-alg>
          1. Let _genKind_ be GetGeneratorKind().
          1. If _genKind_ is ~async~, return ? AsyncGeneratorYield(? Await(_arg_)).
          1. Return ? GeneratorYield(CreateIteratorResultObject(_arg_, *false*)).
        </emu-alg>

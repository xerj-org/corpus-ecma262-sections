<emu-clause id="sec-typedarray-get" oldids="sec-integer-indexed-exotic-objects-get-p-receiver" type="internal method">
        <h1>
          [[Get]] (
            _propertyKey_: a property key,
            _receiver_: an ECMAScript language value,
          ): either a normal completion containing an ECMAScript language value or a throw completion
        </h1>
        <dl class="header">
          <dt>for</dt>
          <dd>a TypedArray _obj_</dd>
        </dl>
        <emu-alg>
          1. If _propertyKey_ is a String, then
            1. Let _numericIndex_ be CanonicalNumericIndexString(_propertyKey_).
            1. If _numericIndex_ is not *undefined*, then
              1. Return TypedArrayGetElement(_obj_, _numericIndex_).
          1. Return ? OrdinaryGet(_obj_, _propertyKey_, _receiver_).
        </emu-alg>
      </emu-clause>

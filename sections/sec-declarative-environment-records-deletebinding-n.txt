<emu-clause id="sec-declarative-environment-records-deletebinding-n" type="concrete method">
          <h1>
            DeleteBinding (
              _name_: a String,
            ): a normal completion containing a Boolean
          </h1>
          <dl class="header">
            <dt>for</dt>
            <dd>a Declarative Environment Record _envRecord_</dd>

            <dt>description</dt>
            <dd>It can only delete bindings that have been explicitly designated as being subject to deletion.</dd>
          </dl>
          <emu-alg>
            1. Assert: _envRecord_ has a binding for _name_.
            1. If the binding for _name_ in _envRecord_ cannot be deleted, return *false*.
            1. Remove the binding for _name_ from _envRecord_.
            1. Return *true*.
          </emu-alg>
        </emu-clause>

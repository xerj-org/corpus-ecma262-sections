<emu-clause id="sec-module-namespace-exotic-objects-hasproperty-p" type="internal method">
        <h1>
          [[HasProperty]] (
            _propertyKey_: a property key,
          ): a normal completion containing a Boolean
        </h1>
        <dl class="header">
          <dt>for</dt>
          <dd>a module namespace exotic object _obj_</dd>
        </dl>
        <emu-alg>
          1. If _propertyKey_ is a Symbol, return ! OrdinaryHasProperty(_obj_, _propertyKey_).
          1. Let _exports_ be _obj_.[[Exports]].
          1. If _exports_ contains _propertyKey_, return *true*.
          1. Return *false*.
        </emu-alg>
      </emu-clause>

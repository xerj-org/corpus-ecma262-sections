<emu-clause id="sec-symboldescriptivestring" type="abstract operation">
          <h1>
            SymbolDescriptiveString (
              _symbol_: a Symbol,
            ): a String
          </h1>
          <dl class="header">
          </dl>
          <emu-alg>
            1. Let _description_ be _symbol_.[[Description]].
            1. If _description_ is *undefined*, set _description_ to the empty String.
            1. Assert: _description_ is a String.
            1. Return the string-concatenation of *"Symbol("*, _description_, and *")"*.
          </emu-alg>
        </emu-clause>

<emu-clause id="sec-function-environment-records-getthisbinding" type="concrete method">
          <h1>GetThisBinding ( ): either a normal completion containing an ECMAScript language value or a throw completion</h1>
          <dl class="header">
            <dt>for</dt>
            <dd>a Function Environment Record _envRecord_</dd>
          </dl>
          <emu-alg>
            1. Assert: _envRecord_.[[ThisBindingStatus]] is not ~lexical~.
            1. If _envRecord_.[[ThisBindingStatus]] is ~uninitialized~, throw a *ReferenceError* exception.
            1. Return _envRecord_.[[ThisValue]].
          </emu-alg>
        </emu-clause>

<emu-clause id="sec-todatestring" type="abstract operation">
          <h1>
            ToDateString (
              _tv_: an integral Number or *NaN*,
            ): a String
          </h1>
          <dl class="header">
          </dl>
          <emu-alg>
            1. If _tv_ is *NaN*, return *"Invalid Date"*.
            1. Let _localTime_ be LocalTime(_tv_).
            1. Return the string-concatenation of DateString(_localTime_), the code unit 0x0020 (SPACE), TimeString(_localTime_), and TimeZoneString(_tv_).
          </emu-alg>
        </emu-clause>

<emu-clause id="sec-thisstringvalue" type="abstract operation" oldids="thisstringvalue">
          <h1>
            ThisStringValue (
              _arg_: an ECMAScript language value,
            ): either a normal completion containing a String or a throw completion
          </h1>
          <dl class="header">
          </dl>
          <emu-alg>
            1. If _arg_ is a String, return _arg_.
            1. If _arg_ is an Object and _arg_ has a [[StringData]] internal slot, then
              1. Let _string_ be _arg_.[[StringData]].
              1. Assert: _string_ is a String.
              1. Return _string_.
            1. Throw a *TypeError* exception.
          </emu-alg>
        </emu-clause>
